cmake_minimum_required(VERSION 3.16)
project(five_tratum_oled_native C)
set(CMAKE_C_STANDARD 11)
get_filename_component(FW_ROOT "../../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
add_link_options(-fsanitize=address,undefined)
add_subdirectory("${FW_ROOT}/managed_components/lvgl__lvgl" lvgl)
add_executable(oled_native oled_native.c "${FW_ROOT}/main/lv_font_portfolio-6x8.c")
target_include_directories(oled_native PRIVATE "${FW_ROOT}/main")
target_compile_options(oled_native PRIVATE -Wall -Wextra -Werror)
target_link_libraries(oled_native PRIVATE lvgl m)
add_executable(oled_screen_runtime oled_screen_runtime.c "${FW_ROOT}/main/lv_font_portfolio-6x8.c")
target_include_directories(oled_screen_runtime PRIVATE "${FW_ROOT}/main" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(oled_screen_runtime PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
target_link_libraries(oled_screen_runtime PRIVATE lvgl m)
